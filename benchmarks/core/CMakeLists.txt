cmake_minimum_required(VERSION 3.20)
project(rabitqlib_core_benchmark LANGUAGES CXX)
if(NOT TARGET rabitqlib::core)
    find_package(rabitqlib CONFIG REQUIRED COMPONENTS core)
endif()
add_executable(rabitq_core_benchmark main.cpp)
target_link_libraries(rabitq_core_benchmark PRIVATE rabitqlib::core)
add_executable(rabitq_core_scoring_benchmark scoring.cpp scoring_legacy.cpp)
target_link_libraries(rabitq_core_scoring_benchmark PRIVATE rabitqlib::core)
# The out-of-line comparator requires a genuine translation-unit boundary.
set_target_properties(rabitq_core_scoring_benchmark PROPERTIES INTERPROCEDURAL_OPTIMIZATION OFF)
if(COMMAND rabitq_configure_target)
    rabitq_configure_target(rabitq_core_benchmark)
    rabitq_configure_target(rabitq_core_scoring_benchmark)
endif()
enable_testing()
add_test(NAME CoreBenchmarkSmoke COMMAND rabitq_core_benchmark 96 4 256 8 1)
add_test(NAME CoreBenchmarkOneBit COMMAND rabitq_core_benchmark 64 1 32 2 1)
add_test(NAME CoreScoringPacked COMMAND rabitq_core_scoring_benchmark 128 4 packed hot 4096 1 1)
add_test(NAME CoreScoringUnpacked COMMAND rabitq_core_scoring_benchmark 65 9 unpacked random 4096 1 1)
add_test(NAME CoreScoringOneBit COMMAND rabitq_core_scoring_benchmark 32 1 packed sequential 4096 1 1)
